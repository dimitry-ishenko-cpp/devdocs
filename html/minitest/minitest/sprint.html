def self.run args = ARGV
  if args.delete("--bisect") or args.delete("-b") then
    require_relative "bisect"

    return Minitest::Bisect.run ARGV
  end

  Minitest::PathExpander.new(args).process { |f|
    require "./#{f}" if File.file? f
  }
end
