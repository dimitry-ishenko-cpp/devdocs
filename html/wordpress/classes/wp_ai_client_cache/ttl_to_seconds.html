private function ttl_to_seconds( $ttl ): int {
	if ( null === $ttl ) {
		return 0;
	}

	if ( $ttl instanceof DateInterval ) {
		$now = new DateTime();
		$end = ( clone $now )->add( $ttl );

		return $end->getTimestamp() - $now->getTimestamp();
	}

	return max( 0, (int) $ttl );
}
