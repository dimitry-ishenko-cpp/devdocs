protected function generate_cache_key( array $deprecated, $sql ) {
	global $wpdb;

	// Replace wpdb placeholder in the SQL statement used by the cache key.
	$sql = $wpdb->remove_placeholder_escape( $sql );

	$key = md5( $sql );

	return "get_users:$key";
}
