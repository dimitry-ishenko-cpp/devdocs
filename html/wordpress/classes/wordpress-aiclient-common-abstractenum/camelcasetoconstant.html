private static function camelCaseToConstant(string $camelCase): string
{
    $snakeCase = preg_replace('/([a-z])([A-Z])/', '$1_$2', $camelCase);
    if ($snakeCase === null) {
        return strtoupper($camelCase);
    }
    return strtoupper($snakeCase);
}
