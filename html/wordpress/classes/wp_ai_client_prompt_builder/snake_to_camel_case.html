private function snake_to_camel_case( string $snake_case ): string {
	$parts = explode( '_', $snake_case );

	$camel_case  = $parts[0];
	$parts_count = count( $parts );
	for ( $i = 1; $i < $parts_count; $i++ ) {
		$camel_case .= ucfirst( $parts[ $i ] );
	}

	return $camel_case;
}
