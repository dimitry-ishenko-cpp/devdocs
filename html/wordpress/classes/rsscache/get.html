function get ($url) {
	$this->ERROR = "";
	$cache_option = 'rss_' . $this->file_name( $url );

	if ( ! $rss = get_transient( $cache_option ) ) {
		$this->debug(
			"Cache does not contain: $url (cache option: $cache_option)"
		);
		return 0;
	}

	return $rss;
}
