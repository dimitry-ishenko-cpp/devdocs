public function __construct( array $contexts = array() ) {
	if ( count( $contexts ) > 0 ) {
		$this->contexts = array_map(
			static function ( string $str ): string {
				return self::escape_pattern_string( trailingslashit( $str ) );
			},
			$contexts
		);
	} else {
		$this->contexts = self::get_default_contexts();
	}
}
