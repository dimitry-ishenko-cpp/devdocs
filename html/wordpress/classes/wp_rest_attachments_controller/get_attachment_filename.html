protected function get_attachment_filename( int $attachment_id ): ?string {
	$path = wp_get_original_image_path( $attachment_id );

	if ( $path ) {
		return wp_basename( $path );
	}

	$path = get_attached_file( $attachment_id );

	if ( $path ) {
		return wp_basename( $path );
	}

	return null;
}
