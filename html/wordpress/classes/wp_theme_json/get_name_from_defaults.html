protected function get_name_from_defaults( $slug, $base_path ) {
	$path            = $base_path;
	$path[]          = 'default';
	$default_content = _wp_array_get( $this->theme_json, $path, null );
	if ( ! $default_content ) {
		return null;
	}
	foreach ( $default_content as $item ) {
		if ( $slug === $item['slug'] ) {
			return $item['name'];
		}
	}
	return null;
}
