public static function fromArray(array $array): self
{
    static::validateFromArrayData($array, [self::KEY_RESPONSE]);
    return new self($array[self::KEY_ID] ?? null, $array[self::KEY_NAME] ?? null, $array[self::KEY_RESPONSE]);
}
