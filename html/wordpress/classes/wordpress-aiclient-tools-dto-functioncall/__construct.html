public function __construct(?string $id = null, ?string $name = null, $args = null)
{
    if ($id === null && $name === null) {
        throw new InvalidArgumentException('At least one of id or name must be provided.');
    }
    $this->id = $id;
    $this->name = $name;
    $this->args = $args;
}
