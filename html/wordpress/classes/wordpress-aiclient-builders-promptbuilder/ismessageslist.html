Source

private function isMessagesList($value): bool
{
    if (!is_array($value) || empty($value) || !array_is_list($value)) {
        return \false;
    }
    // Check if all items are Messages
    foreach ($value as $item) {
        if (!$item instanceof Message) {
            return \false;
        }
    }
    return \true;
}
