private function normalizePreferenceIdentifier($value, string $emptyMessage = 'Model preference identifiers cannot be empty.'): string
{
    if (!is_string($value)) {
        throw new InvalidArgumentException($emptyMessage);
    }
    $trimmed = trim($value);
    if ($trimmed === '') {
        throw new InvalidArgumentException($emptyMessage);
    }
    return $trimmed;
}
