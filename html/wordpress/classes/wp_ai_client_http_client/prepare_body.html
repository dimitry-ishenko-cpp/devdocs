private function prepare_body( RequestInterface $request ): ?string {
	$body = $request->getBody();

	if ( $body->getSize() === 0 ) {
		return null;
	}

	if ( $body->isSeekable() ) {
		$body->rewind();
	}

	return (string) $body;
}
