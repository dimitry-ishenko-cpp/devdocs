public function __construct( $html ) {
	if ( ! is_string( $html ) ) {
		_doing_it_wrong(
			__METHOD__,
			__( 'The HTML parameter must be a string.' ),
			'6.9.0'
		);
		$html = '';
	}
	$this->html = $html;
}
