WP_MS_Themes_List_Table::no_items()

Displays the message when there are no items to list.

Source

public function no_items() {
	if ( $this->has_items ) {
		_e( 'No themes found.' );
	} else {
		_e( 'No themes are currently available.' );
	}
}
