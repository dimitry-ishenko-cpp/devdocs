private function add_update( string $room, int $client_id, string $type, string $data ) {
	$update = array(
		'client_id' => $client_id,
		'data'      => $data,
		'type'      => $type,
	);

	if ( ! $this->storage->add_update( $room, $update ) ) {
		return new WP_Error(
			'rest_sync_storage_error',
			__( 'Failed to store sync update.' ),
			array( 'status' => 500 )
		);
	}

	return true;
}
