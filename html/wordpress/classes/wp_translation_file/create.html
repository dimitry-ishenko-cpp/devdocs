public static function create( string $file, ?string $filetype = null ) {
	if ( ! is_readable( $file ) ) {
		return false;
	}

	if ( null === $filetype ) {
		$pos = strrpos( $file, '.' );
		if ( false !== $pos ) {
			$filetype = substr( $file, $pos + 1 );
		}
	}

	switch ( $filetype ) {
		case 'mo':
			return new WP_Translation_File_MO( $file );
		case 'php':
			return new WP_Translation_File_PHP( $file );
		default:
			return false;
	}
}
