protected function get_plural_expression_from_header( string $header ): string {
	if ( preg_match( '/^\s*nplurals\s*=\s*(\d+)\s*;\s+plural\s*=\s*(.+)$/', $header, $matches ) ) {
		return trim( $matches[2] );
	}

	return 'n != 1';
}
