public function toFiles(): array
{
    $files = [];
    foreach ($this->candidates as $candidate) {
        $message = $candidate->getMessage();
        foreach ($message->getParts() as $part) {
            $channel = $part->getChannel();
            $file = $part->getFile();
            if ($channel->isContent() && $file !== null) {
                $files[] = $file;
                break;
            }
        }
    }
    return $files;
}
