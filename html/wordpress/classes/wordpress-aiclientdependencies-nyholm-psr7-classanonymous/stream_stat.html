public function stream_stat(): array
{
    return [
        'mode' => 33206,
        // POSIX_S_IFREG | 0666
        'nlink' => 1,
        'rdev' => -1,
        'size' => \strlen($this->data),
        'blksize' => -1,
        'blocks' => -1,
    ];
}
