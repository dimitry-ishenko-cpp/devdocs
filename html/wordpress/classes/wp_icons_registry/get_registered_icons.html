public function get_registered_icons( $search = '' ) {
	$icons = array();

	foreach ( $this->registered_icons as $icon ) {
		if ( ! empty( $search ) && false === stripos( $icon['name'], $search ) ) {
			continue;
		}

		$icon['content'] = $icon['content'] ?? $this->get_content( $icon['name'] );
		$icons[]         = $icon;
	}

	return $icons;
}
