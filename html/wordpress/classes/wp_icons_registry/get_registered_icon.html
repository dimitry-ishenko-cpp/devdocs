public function get_registered_icon( $icon_name ) {
	if ( ! $this->is_registered( $icon_name ) ) {
		return null;
	}

	$icon            = $this->registered_icons[ $icon_name ];
	$icon['content'] = $icon['content'] ?? $this->get_content( $icon_name );

	return $icon;
}
