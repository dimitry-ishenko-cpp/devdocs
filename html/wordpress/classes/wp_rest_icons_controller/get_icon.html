public function get_icon( $name ) {
	$registry = WP_Icons_Registry::get_instance();
	$icon     = $registry->get_registered_icon( $name );

	if ( null === $icon ) {
		return new WP_Error(
			'rest_icon_not_found',
			sprintf(
				// translators: %s is the name of any user-provided name
				__( 'Icon not found: "%s".' ),
				$name
			),
			array( 'status' => 404 )
		);
	}

	return $icon;
}
