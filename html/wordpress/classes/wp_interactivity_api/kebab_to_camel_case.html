private function kebab_to_camel_case( string $str ): string {
	return lcfirst(
		preg_replace_callback(
			'/(-)([a-z])/',
			function ( $matches ) {
				return strtoupper( $matches[2] );
			},
			strtolower( rtrim( $str, '-' ) )
		)
	);
}
