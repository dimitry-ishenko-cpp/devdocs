Outputs a tag_name XML tag from a given tag object.

Parameters

Tag Object.

Source

function wxr_tag_name( $tag ) {
	if ( empty( $tag->name ) ) {
		return;
	}

	echo '<wp:tag_name>' . wxr_cdata( $tag->name ) . "</wp:tag_name>\n";
}
