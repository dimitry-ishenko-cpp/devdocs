Outputs a cat_name XML tag from a given category object.

Parameters

$categoryWP_Termrequired
Category Object.

Source

function wxr_cat_name( $category ) {
	if ( empty( $category->name ) ) {
		return;
	}

	echo '<wp:cat_name>' . wxr_cdata( $category->name ) . "</wp:cat_name>\n";
}
