TrailingSlashPathLocationStrategy

A LocationStrategy that ensures URLs always have a trailing slash. This strategy only affects the URL written to the browser. Location.path() and Location.normalize() will continue to strip trailing slashes when reading the URL.

API

class TrailingSlashPathLocationStrategy extends PathLocationStrategy {
  prepareExternalUrl(internal: string): string;
  override onPopState(fn: LocationChangeListener): void;
  override getBaseHref(): string;
  override path(includeHash?: boolean): string;
  override pushState(state: any, title: string, url: string, queryParams: string): void;
  override replaceState(state: any, title: string, url: string, queryParams: string): void;
  override forward(): void;
  override back(): void;
  override getState(): unknown;
  override historyGo(relativePosition?: number): void;
}
