reduceRightOrNull

kotlin-stdlib/kotlin.text/reduceRightOrNull

Since Kotlin: 1.4

inline fun CharSequence.reduceRightOrNull(operation: (Char, acc: Char) -> Char): Char?

Accumulates value starting with the last character and applying operation from right to left to each character and current accumulator value.

Returns null if the char sequence is empty.

Parameters

operation

function that takes a character and current accumulator value, and calculates the next accumulator value.

Samples

import kotlin.math.*
import kotlin.test.*

fun main() { 
   //sampleStart 
   val strings = listOf("a", "b", "c", "d")
println(strings.reduceRightOrNull { string, acc -> acc + string }) // dcba
println(strings.reduceRightIndexedOrNull { index, string, acc -> acc + string + index }) // dc2b1a0

println(emptyList<String>().reduceRightOrNull { _, _ -> "" }) // null 
   //sampleEnd
