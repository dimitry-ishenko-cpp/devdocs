isBlank

kotlin-stdlib/kotlin.text/isBlank

Since Kotlin: 1.0

fun CharSequence.isBlank(): Boolean

Returns true if this char sequence is empty or consists solely of whitespace characters according to Char.isWhitespace.

Samples

import kotlin.test.*

fun main() { 
   //sampleStart 
   fun validateName(name: String): String {
    if (name.isBlank()) throw IllegalArgumentException("Name cannot be blank")
    return name
}

println(validateName("Adam")) // Adam
// validateName("") // will fail
// validateName("  \t\n") // will fail 
   //sampleEnd
}
