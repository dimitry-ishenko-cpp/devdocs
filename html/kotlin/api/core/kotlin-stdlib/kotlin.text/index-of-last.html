indexOfLast

kotlin-stdlib/kotlin.text/indexOfLast

Since Kotlin: 1.0

inline fun CharSequence.indexOfLast(predicate: (Char) -> Boolean): Int

Returns index of the last character matching the given predicate, or -1 if the char sequence does not contain such character.

Samples

import kotlin.test.*

fun main() { 
   //sampleStart 
   val string = "Kotlin knowledge"

println(string.indexOfLast { it == 'K' }) // 0
println(string.indexOfLast { it == 'k' }) // 7
println(string.indexOfLast { it == 'n' }) // 8
println(string.indexOfLast { it == ' ' }) // 6
println(string.indexOfLast { it == 'z' }) // -1

val emptyString = "";
println(emptyString.indexOfLast { it == 'z' }) // -1 
   //sampleEnd
}
