lastIndexOf

kotlin-stdlib/kotlin.text/lastIndexOf

Since Kotlin: 1.0

fun CharSequence.lastIndexOf(char: Char, startIndex: Int = lastIndex, ignoreCase: Boolean = false): Int

Returns the index within this char sequence of the last occurrence of the specified character, starting from the specified startIndex.

Return

An index of the last occurrence of char or -1 if none is found.

Parameters

startIndex

The index of character to start searching at. The search proceeds backward toward the beginning of the string.

ignoreCase

true to ignore character case when matching a character. By default false.

fun CharSequence.lastIndexOf(string: String, startIndex: Int = lastIndex, ignoreCase: Boolean = false): Int

Returns the index within this char sequence of the last occurrence of the specified string, starting from the specified startIndex.

Return

An index of the last occurrence of string or -1 if none is found.

Parameters

startIndex

The index of character to start searching at. The search proceeds backward toward the beginning of the string.

ignoreCase

true to ignore character case when matching a string. By default false.

import kotlin.test.*

   fun matchDetails(inputString: String, whatToFind: String, startIndex: Int = inputString.length - 1): String {
    val matchIndex = inputString.lastIndexOf(whatToFind, startIndex)
    return "Searching for the last '$whatToFind' in '$inputString' starting at position $startIndex: " +
            if (matchIndex >= 0) "Found at $matchIndex" else "Not found"
}

val inputString = "Never ever give up"
val toFind = "ever"

println(matchDetails(inputString, toFind)) // Searching for the last 'ever' in 'Never ever give up' starting at position 17: Found at 6
println(matchDetails(inputString, toFind, 0)) // Searching for the last 'ever' in 'Never ever give up' starting at position 0: Not found
println(matchDetails(inputString, toFind, 5)) // Searching for the last 'ever' in 'Never ever give up' starting at position 5: Found at 1 
   //sampleEnd
}
