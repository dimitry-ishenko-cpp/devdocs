takeWhile

kotlin-stdlib/kotlin.text/takeWhile

Since Kotlin: 1.0

inline fun CharSequence.takeWhile(predicate: (Char) -> Boolean): CharSequence

Returns a subsequence of this char sequence containing the first characters that satisfy the given predicate.

Samples

import kotlin.test.*

fun main() { 
   //sampleStart 
   val string = "<<<First Grade>>>"
println(string.take(8)) // <<<First
println(string.takeLast(8)) // Grade>>>
println(string.takeWhile { !it.isLetter() }) // <<<
println(string.takeLastWhile { !it.isLetter() }) // >>> 
   //sampleEnd
}
