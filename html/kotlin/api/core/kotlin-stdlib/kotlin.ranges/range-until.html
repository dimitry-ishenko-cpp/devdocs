rangeUntil

kotlin-stdlib/kotlin.ranges/rangeUntil

Since Kotlin: 1.9

operator fun <T : Comparable<T>> T.rangeUntil(that: T): OpenEndRange<T>

Creates an open-ended range from this Comparable value to the specified that value.

This value needs to be smaller than that value, otherwise the returned range will be empty.

Samples

import java.sql.Date
import kotlin.test.assertFalse
import kotlin.test.assertTrue

fun main() { 
   //sampleStart 
   val start = Date.valueOf("2017-01-01")
val end = Date.valueOf("2017-12-31")
val range = start..<end
println(range) // 2017-01-01..<2017-12-31

println("Date.valueOf(\"2017-05-27\") in range is ${Date.valueOf("2017-05-27") in range}") // true
println("Date.valueOf(\"2017-12-31\") in range is ${Date.valueOf("2017-12-31") in range}") // false
println("Date.valueOf(\"2017-12-31\") !in range is ${Date.valueOf("2017-12-31") !in range}") // true 
   //sampleEnd
}
operator fun Double.rangeUntil(that: Double): OpenEndRange<Double>

Creates an open-ended range from this Double value to the specified that value.

Numbers are compared with the ends of this range according to IEEE-754.

Samples

import java.sql.Date
import kotlin.test.assertFalse
import kotlin.test.assertTrue

fun main() { 
   //sampleStart 
   val range = 1.0..<100.0
println(range) // 1.0..<100.0

println("3.14 in range is ${3.14 in range}") // true
println("100.0 in range is ${100.0 in range}") // false 
   //sampleEnd
}
