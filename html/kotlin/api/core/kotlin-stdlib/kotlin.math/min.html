min

kotlin-stdlib/kotlin.math/min

Platform and version requirements: Common (1.5)

inline fun min(a: UInt, b: UInt): UInt
inline fun min(a: ULong, b: ULong): ULong

Returns the smaller of two values.

expect fun min(a: Double, b: Double): Double

Returns the smaller of two values.

If either value is NaN, then the result is NaN.

Samples

import kotlin.math.*
import kotlin.test.*

fun main() { 
   //sampleStart 
   println(min(-1.0, 1.0)) // -1.0
println(min(1.0, -1.0)) // -1.0
println(min(-3.0, -1.0)) // -3.0
println(min(-0.0, 0.0)) // -0.0

// special cases
println(min(Double.NaN, 0.0)) // NaN
println(min(-9000.0, Double.NaN)) // NaN
// Note that MIN_VALUE has a different meaning compared to Long.MIN_VALUE or Int.MIN_VALUE
println(min(Double.MIN_VALUE, 0.0)) // 0.0
println("min(Double.POSITIVE_INFINITY, Double.MAX_VALUE) == Double.MAX_VALUE is ${min(Double.POSITIVE_INFINITY, Double.MAX_VALUE) == Double.MAX_VALUE}") // true 
   //sampleEnd
}
