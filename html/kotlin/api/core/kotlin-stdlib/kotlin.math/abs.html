abs

kotlin-stdlib/kotlin.math/abs

Platform and version requirements: Common (1.2)

expect fun abs(x: Double): Double

Returns the absolute value of the given value x.

Special cases:

  • abs(NaN) is NaN

See also

absoluteValue

extension property for Double

Samples

import kotlin.math.*
import kotlin.test.*

fun main() { 
   //sampleStart 
   println(abs(3.14)) // 3.14
println(abs(-3.14)) // 3.14
println(abs(-0.0)) // 0.0
println(abs(Double.NEGATIVE_INFINITY)) // Infinity
println(abs(Double.NaN)) // NaN 
   //sampleEnd
}
expect fun abs(x: Float): Float

Returns the absolute value of the given value x.

Special cases:

See also

absoluteValue

extension property for Float

Samples

import kotlin.math.*
import kotlin.test.*

   //sampleStart

println(Float.NaN.absoluteValue) // NaN 
   //sampleEnd
}
expect fun abs(n: Int): Int

Returns the absolute value of the given value n.

Special cases:

  • abs(Int.MIN_VALUE) is Int.MIN_VALUE due to an overflow
