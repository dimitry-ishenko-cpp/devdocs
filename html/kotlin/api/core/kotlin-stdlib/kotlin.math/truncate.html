truncate

kotlin-stdlib/kotlin.math/truncate

Platform and version requirements: Common (1.2), JVM (1.2), Native (1.3), Wasm-JS (1.8), Wasm-WASI (1.8)

expect fun truncate(x: Double): Double

Rounds the given value x to an integer towards zero.

Special cases:

  • truncate(x) is x where x is NaN or +Inf or -Inf or already a mathematical integer.

Return

the value x having its fractional part truncated.

Samples

import kotlin.math.*
import kotlin.test.*

fun main() { 
   //sampleStart 
   println(truncate(3.14159)) // 3.0
println(truncate(-1.1)) // -1.0
// 10.0 is already an "integer", so no rounding will take place
println(truncate(10.0)) // 10.0
// Special cases
println(truncate(Double.NaN)) // NaN
println(truncate(Double.POSITIVE_INFINITY)) // Infinity
println(truncate(Double.NEGATIVE_INFINITY)) // -Infinity 
   //sampleEnd
}
import kotlin.math.*
import kotlin.test.*

fun main() { 
   //sampleStart 
   println(floor(3.5)) // 3.0
println(ceil(3.5)) // 4.0
println(truncate(3.5)) // 3.0
println(round(3.5)) // 4.0
println(round(3.49)) // 3.0

println(floor(-3.5)) // -4.0
println(ceil(-3.5)) // -3.0
println(truncate(-3.5)) // -3.0
println(round(-3.5)) // -4.0
println(round(-3.49)) // -3.0 
   //sampleEnd
}
