booleanArrayOf

kotlin-stdlib/kotlin/booleanArrayOf

Platform and version requirements: Common (1.0), JVM (1.0)

expect fun booleanArrayOf(vararg elements: Boolean): BooleanArray

Returns an array containing the specified boolean values.

Samples

import kotlin.test.*

fun main() { 
   //sampleStart 
   val emptyBooleanArray = booleanArrayOf()
println(emptyBooleanArray.contentToString()) // []

val booleanArray = booleanArrayOf(true, false, true)
println(booleanArray.contentToString()) // [true, false, true] 
   //sampleEnd
}
