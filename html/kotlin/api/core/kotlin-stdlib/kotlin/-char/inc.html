inc

kotlin-stdlib/kotlin/Char/inc

Platform and version requirements: Common (1.0), JS (1.1), JVM (1.0)

expect operator fun inc(): Char

Returns this value incremented by one.

Samples

import kotlin.test.assertFailsWith

fun main() { 
   //sampleStart 
   val a = 3
val b = a.inc()
println(a) // 3
println(b) // 4

var x = 3
val y = x++
println(x) // 4
println(y) // 3

val z = ++x
println(x) // 5
println(z) // 5 
   //sampleEnd
}
