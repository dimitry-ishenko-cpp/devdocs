byteArrayOf

kotlin-stdlib/kotlin/byteArrayOf

Platform and version requirements: Common (1.0), JVM (1.0)

expect fun byteArrayOf(vararg elements: Byte): ByteArray

Returns an array containing the specified Byte numbers.

Samples

import kotlin.test.*

fun main() { 
   //sampleStart 
   val emptyByteArray = byteArrayOf()
println(emptyByteArray.contentToString()) // []

val byteArray = byteArrayOf(1, 2, 3)
println(byteArray.contentToString()) // [1, 2, 3] 
   //sampleEnd
}
