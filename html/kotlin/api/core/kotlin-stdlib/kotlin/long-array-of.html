longArrayOf

kotlin-stdlib/kotlin/longArrayOf

Platform and version requirements: Common (1.0), JVM (1.0)

expect fun longArrayOf(vararg elements: Long): LongArray

Returns an array containing the specified Long numbers.

Samples

import kotlin.test.*

fun main() { 
   //sampleStart 
   val emptyLongArray = longArrayOf()
println(emptyLongArray.contentToString()) // []

val longArray = longArrayOf(1L, 2L, 3L)
println(longArray.contentToString()) // [1, 2, 3] 
   //sampleEnd
}
