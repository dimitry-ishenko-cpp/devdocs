inc

kotlin-stdlib/kotlin/UShort/inc

Since Kotlin: 1.5

inline operator fun inc(): UShort

Returns this value incremented by one.

Samples

import kotlin.test.assertFailsWith

fun main() { 
   //sampleStart 
   val a = 3
val b = a.inc()
println(a) // 3
println(b) // 4

var x = 3
val y = x++
println(x) // 4
println(y) // 3

val z = ++x
println(x) // 5
println(z) // 5 
   //sampleEnd
}
