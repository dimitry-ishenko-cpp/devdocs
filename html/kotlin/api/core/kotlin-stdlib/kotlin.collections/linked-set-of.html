inline fun <T> linkedSetOf(): LinkedHashSet<T>

Returns an empty new LinkedHashSet.

Samples

import kotlin.math.*
import kotlin.test.*

fun main() { 
   //sampleStart 
   val set: LinkedHashSet<Int> = linkedSetOf<Int>()

set.add(1)
set.add(3)
set.add(2)

println(set) // [1, 3, 2] 
   //sampleEnd
}
fun <T> linkedSetOf(vararg elements: T): LinkedHashSet<T>

Returns a new LinkedHashSet with the given elements. Elements of the set are iterated in the order they were specified.

Samples

import kotlin.math.*
import kotlin.test.*

fun main() { 
   //sampleStart 
   val set: LinkedHashSet<Int> = linkedSetOf(1, 3, 2)

println(set) // [1, 3, 2]

set.remove(3)
set += listOf(5, 4)
println(set) // [1, 2, 5, 4] 
   //sampleEnd
}
