Returns a string representation of the contents of the specified array as if it is List.

Samples

import kotlin.test.*

fun main() { 
   //sampleStart 
   val array = arrayOf("apples", "oranges", "lime")

println(array.contentToString()) // [apples, oranges, lime] 
   //sampleEnd
}
