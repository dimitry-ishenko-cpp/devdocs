toDurationUnit

kotlin-stdlib/kotlin.time/toDurationUnit

Since Kotlin: 1.8

fun TimeUnit.toDurationUnit(): DurationUnit

Converts this java.util.concurrent.TimeUnit enum value to the corresponding kotlin.time.DurationUnit value.

Samples

import kotlin.test.*
import kotlin.time.*
import kotlin.time.Duration.Companion.days
import kotlin.time.Duration.Companion.hours
import kotlin.time.Duration.Companion.microseconds
import kotlin.time.Duration.Companion.milliseconds
import kotlin.time.Duration.Companion.minutes
import kotlin.time.Duration.Companion.nanoseconds
import kotlin.time.Duration.Companion.seconds

fun main() { 
   //sampleStart 
   val durationUnit: DurationUnit = java.util.concurrent.TimeUnit.MILLISECONDS.toDurationUnit()
println(durationUnit) // MILLISECONDS 
   //sampleEnd
}
