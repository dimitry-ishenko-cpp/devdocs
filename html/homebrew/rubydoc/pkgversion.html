def self.parse(path)
  _, version, revision = *path.match(REGEX)
  version = Version.new(version.to_s)
  new(version, revision.to_i)
end

def to_str
  if revision.positive?
    "#{version}_#{revision}"
  else
    version.to_s
  end
end
