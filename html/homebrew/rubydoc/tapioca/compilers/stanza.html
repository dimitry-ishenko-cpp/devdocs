def decorate
  root.create_module(T.must(constant.name)) do |mod|
    ::RuboCop::Cask::Constants::STANZA_ORDER.each do |stanza|
      mod.create_method("#{stanza}?", return_type: "T::Boolean", class_method: false)
    end
  end
end
