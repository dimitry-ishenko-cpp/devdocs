def run
  formulae = Homebrew::Services::Formulae.services_list
  if formulae.blank?
    opoo "No services available to control with `#{Homebrew::Services::Cli.bin}`" if $stderr.tty?
    puts "[]" if args.json?
    return
  end

  if args.json?
    self.class.print_json(formulae)
  else
    self.class.print_table(formulae)
  end
end
