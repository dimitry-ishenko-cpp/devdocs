def self.create(package_or_resource, version)
  versions = case package_or_resource
  when Formula, Resource
    [version]
  when Cask::Cask
    version.to_s.split(",").map { |s| Version.new(s) }
  else
    T.absurd(package_or_resource)
  end
  new(versions)
end
