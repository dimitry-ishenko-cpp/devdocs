def self.checks(type, fatal: true)
  @checks ||= T.let(Checks.new, T.nilable(Checks))
  failed = T.let(false, T::Boolean)
  @checks.public_send(type).each do |check|
    out = @checks.public_send(check)
    next if out.nil?

    if fatal
      failed ||= true
      ofail out
    else
      opoo out
    end
  end
  exit 1 if failed && fatal
end
