def env(*settings)
  env = @env
  Kernel.raise ArgumentError, "#{self} has not been initialized with a BuildEnvironment" if env.nil?

  env.merge(settings)
end

def inherited(child)
  super
  child.instance_eval do
    @env = T.let(BuildEnvironment.new, T.nilable(BuildEnvironment))
  end
end
