def initialize(token, loaders)
  @token = token
  @loaders = loaders

  taps = loaders.map(&:tap)
  casks = taps.map { |tap| "#{tap}/#{token}" }
  cask_list = casks.sort.map { |f| "\n       * #{f}" }.join

  super <<~EOS
    Cask #{token} exists in multiple taps:#{cask_list}    Please use the fully-qualified name (e.g. #{casks.first}) to refer to a specific Cask.
  EOS
end
