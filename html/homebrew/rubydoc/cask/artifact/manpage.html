def self.from_args(cask, source, _target_hash = nil)
  section = source.to_s[/\.([1-8]|n|l)(?:\.gz)?$/, 1]

  raise CaskInvalidError, "'#{source}' is not a valid man page name" unless section

  new(cask, source, section)
end
