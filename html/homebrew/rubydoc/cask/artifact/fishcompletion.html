def resolve_target(target, base_dir: nil)
  name = if target.to_s.end_with? ".fish"
    target
  else
    new_name = "#{File.basename(target, File.extname(target))}.fish"
    odebug "Renaming completion #{target} to #{new_name}"

    new_name
  end

  config.fish_completion/name
end
