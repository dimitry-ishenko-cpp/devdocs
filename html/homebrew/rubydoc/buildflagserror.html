def initialize(flags, bottled: true)
  if flags.length > 1
    flag_text = "flags"
    require_text = "require"
  else
    flag_text = "flag"
    require_text = "requires"
  end

  bottle_text = if bottled
    <<~EOS
      Alternatively, remove the #{flag_text} to attempt bottle installation.
    EOS
  end

  message = <<~EOS
    The following #{flag_text}:
#{flags.join(", ")}#{require_text} building tools, but none are installed.
#{DevelopmentTools.installation_instructions}#{bottle_text}  EOS

  super message
end
