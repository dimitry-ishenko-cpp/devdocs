def apply
  data = contents.gsub("@@HOMEBREW_PREFIX@@", HOMEBREW_PREFIX)
  args = %W[-g0-f-#{strip}]
  dir = Pathname.pwd
  if (subdirectory = directory.presence)
    dir /= subdirectory
  end
  dir.cd do
    Utils.safe_popen_write("patch", *args) { |p| p.write(data) }
  end
end
