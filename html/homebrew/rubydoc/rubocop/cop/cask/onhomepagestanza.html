def on_cask(cask_block)
  @cask_block = T.let(cask_block, T.nilable(RuboCop::Cask::AST::CaskBlock))

  toplevel_stanzas.select(&:homepage?).each do |stanza|
    on_homepage_stanza(stanza)
  end
end
