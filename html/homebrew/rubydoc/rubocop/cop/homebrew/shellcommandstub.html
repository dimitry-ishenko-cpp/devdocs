def on_send(node)
  return if node.first_argument&.const_name != "ShellCommand"

  stub_path = Pathname.new(processed_source.file_path)
  sh_cmd_path = Pathname.new("#{stub_path.dirname}/#{stub_path.basename(".rb")}.sh")
  return if sh_cmd_path.exist?

  add_offense(node)
end
