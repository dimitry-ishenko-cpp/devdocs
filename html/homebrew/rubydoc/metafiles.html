def copy?(file)
  file = file.downcase
  license = file.split(/\.|-/).first
  return false unless license
  return true if LICENSES.include?(license)

  ext  = File.extname(file)
  file = File.basename(file, ext) if EXTENSIONS.include?(ext)
  BASENAMES.include?(file)
end
