def os
  @os ||= T.let(nil, T.nilable(Symbol))
  if @os.blank? && Homebrew::EnvConfig.simulate_macos_on_linux?
    return MacOSVersion.new(HOMEBREW_MACOS_NEWEST_SUPPORTED).to_sym
  end

  @os
end
