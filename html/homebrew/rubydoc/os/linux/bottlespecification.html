def skip_relocation?(tag: Utils::Bottles.tag, tab: nil)
  # Homebrew versions prior to 5.1.15 generated incorrect :any_skip_relocation
  !tab.nil? && tab.parsed_homebrew_version >= "5.1.15" && super
end
