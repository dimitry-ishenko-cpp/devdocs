def std_cmake_args(install_prefix: prefix, install_libdir: "lib", find_framework: "LAST")
  args = super

  # Ensure CMake is using the same SDK we are using.
  args << "-DCMAKE_OSX_SYSROOT=#{T.must(MacOS.sdk_for_formula(self)).path}"

  args
end
