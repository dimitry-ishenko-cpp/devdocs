def nested_sandbox?
  sandbox_check = Fiddle::Function.new(
    Fiddle.dlopen(nil)["sandbox_check"],
    [Fiddle::TYPE_INT, Fiddle::TYPE_VOIDP, Fiddle::TYPE_INT],
    Fiddle::TYPE_INT,
  )
  sandbox_check.call(Process.pid, nil, 0) == 1
rescue Fiddle::DLError
  false
end
