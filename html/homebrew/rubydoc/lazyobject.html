def initialize(&callable)
  @__callable__ = T.let(nil, T.nilable(Proc))
  @getobj_set = T.let(false, T::Boolean)
  @__getobj__ = T.let(nil, T.untyped)
  super(callable)
end

def __getobj__(&_blk)
  return @__getobj__ if @getobj_set

  @__getobj__ = T.must(@__callable__).call
  @getobj_set = true
  @__getobj__
end
