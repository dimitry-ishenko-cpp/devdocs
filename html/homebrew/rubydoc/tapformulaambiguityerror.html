def initialize(name, loaders)
  @name = name
  @loaders = loaders
  @taps = T.let(loaders.filter_map(&:tap), T::Array[Tap])

  formulae = taps.map { |tap| "#{tap}/#{name}" }
  formula_list = formulae.map { |f| "\n       * #{f}" }.join

  super <<~EOS
    Formulae found in multiple taps:#{formula_list}    Please use the fully-qualified name (e.g. #{formulae.first}) to refer to a specific formula.
  EOS
end
