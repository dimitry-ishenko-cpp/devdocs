def initialize(path)
  super(:UNIX, :STREAM)
  bind(UNIXSocketExt.sockaddr_un(path))
  listen(Socket::SOMAXCONN)
  @path = path
end

def accept_nonblock
  socket, = super
  socket.autoclose = false # Transfer autoclose responsibility to UNIXSocket
  UNIXSocket.for_fd(socket.fileno)
end
