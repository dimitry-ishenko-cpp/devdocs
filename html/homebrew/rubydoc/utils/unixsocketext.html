def self.open(path, &_block)
  socket = Socket.new(:UNIX, :STREAM)
  socket.connect(sockaddr_un(path))
  unix_socket = UNIXSocket.for_fd(socket.fileno)
  socket.autoclose = false # Transfer autoclose responsibility to UNIXSocket
  yield unix_socket
end
