static void instantSync(String name, {Map? arguments}) {
  if (!_hasTimeline) return;
  if (!_isDartStreamEnabled()) {
    // Stream is disabled.
    return;
  }
  // Instant events don't have an id because they don't need to be paired with
  // other events.
  int taskId = 0;
  _reportTaskEvent(
    taskId,
    /*flowId=*/ _noFlowId,
    _instant,
    name,
    _argumentsAsJson(arguments),
  );
}
