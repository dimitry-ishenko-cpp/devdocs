Module Result.Syntax

module Syntax: sig .. end

Binding operators.

  • Since 5.4
val ( let* ) : ('a, 'e) result ->       ('a -> ('b, 'e) result) -> ('b, 'e) result

( let* ) is Result.bind.

val ( and* ) : ('a, 'e) result ->       ('b, 'e) result -> ('a * 'b, 'e) result

( and* ) is Result.product.

val let+ : ('a, 'e) result -> ('a -> 'b) -> ('b, 'e) result

( let+ ) is Result.map.

val and+ : ('a, 'e) result ->       ('b, 'e) result -> ('a * 'b, 'e) result

( and+ ) is Result.product.
