class Net::HTTP::Copy

Parent:
Net::HTTPRequest

Class for representing WebDAV method COPY:

require 'net/http'
uri = URI('http://example.com')
hostname = uri.hostname # => "example.com"
req = Net::HTTP::Copy.new(uri) # => #<Net::HTTP::Copy COPY>
res = Net::HTTP.start(hostname) do |http|
  http.request(req)
end

See Request Headers.
