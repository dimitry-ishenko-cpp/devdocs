Standard library header <locale.h>

This header is part of the localization library.

Types
lconv
 formatting details, returned by localeconv
(struct)

Constants

 implementation-defined null pointer constant
(macro constant)

 locale categories for setlocale

Functions
setlocale
 gets and sets the current C locale
(function)

 queries numeric and monetary formatting details of the current locale
(function)

Synopsis

// In the "C" locale, the members shall have the values specified in the comments:
struct lconv
{
    char* decimal_point;        // "."
    char* thousands_sep;        // ""
    char* grouping;             // ""
    char* mon_decimal_point;    // ""
    char* mon_thousands_sep;    // ""
    char* mon_grouping;         // ""
    char* positive_sign;        // ""
    char* negative_sign;        // ""
    char* currency_symbol;      // ""
    char  frac_digits;          // CHAR_MAX
    char  p_cs_precedes;        // CHAR_MAX
    char  n_cs_precedes;        // CHAR_MAX
    char  p_sep_by_space;       // CHAR_MAX
    char  n_sep_by_space;       // CHAR_MAX
    char  p_sign_posn;          // CHAR_MAX
    char  n_sign_posn;          // CHAR_MAX
    char* int_curr_symbol;      // ""
    char  int_frac_digits;      // CHAR_MAX
    char  int_p_cs_precedes;    // CHAR_MAX
    char  int_n_cs_precedes;    // CHAR_MAX
    char  int_p_sep_by_space;   // CHAR_MAX
    char  int_n_sep_by_space;   // CHAR_MAX
    char  int_p_sign_posn;      // CHAR_MAX
    char  int_n_sign_posn;      // CHAR_MAX
};

char* setlocale(int category, const char* locale);
lconv* localeconv();
