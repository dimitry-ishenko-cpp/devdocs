std::swap(std::hive)

Defined in header <hive> 
void swap( std::hive& lhs, std::hive& rhs ) noexcept(/* see below */);
(since C++26)

Specializes the std::swap algorithm for std::hive. Swaps the contents of lhs and rhs. Calls lhs.swap(rhs).

Parameters

lhs, rhs - containers whose contents to swap

Complexity

Constant.

Exceptions

noexcept specification: 
noexcept(noexcept(lhs.swap(rhs)))

Example

#include <algorithm>
#include <iostream>
#include <hive>

int main()
{
    std::hive<int> alice{1, 2, 3};
    std::hive<int> bob{7, 8, 9, 10};

    auto print = [](const int& n) { std::cout << ' ' << n; };

    // Print state before swap
    std::cout << "Alice:";
    std::for_each(alice.begin(), alice.end(), print);
    std::cout << "\nBobby:";
    std::for_each(bob.begin(), bob.end(), print);
    std::cout << '\n';

    std::cout << "-- SWAP\n";
    std::swap(alice, bob);

    // Print state after swap
    std::cout << "Alice:";
    std::for_each(alice.begin(), alice.end(), print);
    std::cout << "\nBobby:";
    std::for_each(bob.begin(), bob.end(), print);
    std::cout << '\n';
}

Output:

Alice: 1 2 3
Bobby: 7 8 9 10
-- SWAP
Alice: 7 8 9 10
Bobby: 1 2 3
