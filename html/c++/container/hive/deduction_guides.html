deduction guides for std::hive 

template< class InputIt,
          class Alloc = std::allocator</*iter-value-type*/<InputIt>> >
hive( InputIt, InputIt, Alloc = Alloc() )
    -> hive</*iter-value-type*/<InputIt>, Alloc>;
(1) (since C++26) 
template< class InputIt,
          class Alloc = std::allocator</*iter-value-type*/<InputIt>> >
hive( InputIt, InputIt, std::hive_limits, Alloc = Alloc() )
    -> hive</*iter-value-type*/<InputIt>, Alloc>;
(2) (since C++26) 
template< ranges::input_range R,
          class Alloc = std::allocator<ranges::range_value_t<R>> >
hive( std::from_range_t, R&&, Alloc = Alloc() )
    -> hive<ranges::range_value_t<R>, Alloc>;
(3) (since C++26) 
template< ranges::input_range R,
          class Alloc = std::allocator<ranges::range_value_t<R>> >
hive( std::from_range_t, R&&, std::hive_limits, Alloc = Alloc() )
    -> hive<ranges::range_value_t<R>, Alloc>;
(4) (since C++26)
1,2) This deduction guide is provided for hive to allow deduction from an iterator range. This overload participates in overload resolution only if InputIt satisfies LegacyInputIterator and Alloc satisfies Allocator.
3,4) This deduction guide is provided for hive to allow deduction from a std::from_range_t tag and an input_range.

Note: the extent to which the library determines that a type does not satisfy LegacyInputIterator is unspecified, except that as a minimum integral types do not qualify as input iterators. Likewise, the extent to which it determines that a type does not satisfy Allocator is unspecified, except that as a minimum the member type Alloc::value_type must exist and the expression std::declval<Alloc&>().allocate(std::size_t{}) must be well-formed when treated as an unevaluated operand.

Example

#include <hive>
#include <list>

int main()
{
    std::list v{1, 2, 3, 4};

    // uses explicit deduction guide to deduce std::hive<int>
    std::hive x(v.begin(), v.end());

    // deduces std::hive<std::vector<int>::iterator>
    // first phase of overload resolution for list-initialization selects
    // the candidate synthesized from the initializer-list constructor;
    // second phase is not performed and deduction guide has no effect
    std::hive y{v.begin(), v.end()};
}
