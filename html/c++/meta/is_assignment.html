std::meta::is_assignment, std::meta::is_copy_assignment, std::meta::is_move_assignment

Defined in header <meta> 
consteval bool is_assignment( std::meta::info r );
(1) (since C++26) 
consteval bool is_copy_assignment( std::meta::info r );
(2) (since C++26) 
consteval bool is_move_assignment( std::meta::info r );
(3) (since C++26)

Returns true if r represents:

1) an assignment operator,
2) a copy assignment operator,
3) a move assignment operator.

Otherwise returns false.

Parameters

r - a reflection value

Return value

true if r represents an assignment operator, otherwise false

Example

#include <iostream>
#include <meta>

template <typename C>
    requires(std::meta::is_class_type(^^C))
constexpr void print_assignment_operators()
{
    constexpr static auto class_name =
        std::define_static_string(std::meta::display_string_of(^^C));
    std::cout << "class " << class_name << '\n';

    constexpr auto ctx = std::meta::access_context::unchecked();
    constexpr static auto members =
        std::define_static_array(std::meta::members_of(^^C, ctx));
    template for (constexpr auto member : members)
    {
        if constexpr (std::meta::is_assignment(member))
        {
            std::cout << "  " << std::meta::display_string_of(member) << " is ";
            if constexpr (std::meta::is_copy_assignment(member))
                std::cout << "copy ";
            else if constexpr (std::meta::is_move_assignment(member))
                std::cout << "move ";
            std::cout << "assignment operator\n";
        }
    }
}

class C
{
    C& operator=(const C&);
    C& operator=(C&&) = default;
    C& operator+=(int);
};

int main()
{
    print_assignment_operators<C>();
}

Possible output:

class C
  C& C::operator=(const C&) is copy assignment operator
  constexpr C& C::operator=(C&&) is move assignment operator
